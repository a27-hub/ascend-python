import os

import numpy as np
import torch_npu
from torch_npu.testing.testcase import TestCase, run_tests

os.environ['PYTORCH_NPU_ALLOC_CONF'] = 'expandable_segments:False'


class TestToPreserveFormatNonContiguous(TestCase):
    """
    测试 tensor.to() 在 memory_format=torch.preserve_format 模式下，
    对各类非连续输入在 D2D / D2H / H2D 场景中的行为。

    核心逻辑（ToKernelNpu.cpp）：
      if (memory_format == Preserve) {
          if (self.is_non_overlapping_and_dense()) {
              // 分支A: 保留所有 strides，用 empty_strided 创建
          } else {
              // 分支B: 降级为 suggest_memory_format()，用 empty 创建
          }
      }
    """

    # ================================================================
    # 辅助方法
    # ================================================================

    def _make_non_contiguous_tensors(self):
        """
        生成各种类型的非连续 tensor，返回列表:
          (name, cpu_tensor, is_non_overlapping_and_dense)
        """
        tensors = []

        # --- 分支A: is_non_overlapping_and_dense = True ---

        # 1. transpose 2D
        t = torch.randn(4, 6).t()
        tensors.append(("transpose_2d", t, True))

        # 2. permute 4D
        t = torch.randn(2, 3, 4, 5).permute(3, 1, 0, 2)
        tensors.append(("permute_4d", t, True))

        # 3. slice dim0 (步长>1)
        t = torch.randn(8, 5)[::2]
        tensors.append(("slice_dim0", t, True))

        # 4. slice dim1 (步长>1)
        t = torch.randn(4, 8)[:, ::2]
        tensors.append(("slice_dim1", t, True))

        # 5. narrow
        t = torch.randn(6, 8).narrow(1, 1, 5)
        tensors.append(("narrow", t, True))

        # 6. select 降维
        t = torch.randn(3, 5, 4).select(1, 2)
        tensors.append(("select_dim1", t, True))

        # 7. 1D 非连续 (as_strided stride>1)
        t = torch.randn(10).as_strided((4,), (2,))
        tensors.append(("1d_as_strided", t, True))

        # --- 分支B: is_non_overlapping_and_dense = False ---

        # 8. expand 2D (stride=0)
        t = torch.randn(1, 5).expand(4, 5)
        tensors.append(("expand_2d", t, False))

        # 9. expand 3D (stride=0)
        t = torch.randn(2, 1, 4).expand(2, 3, 4)
        tensors.append(("expand_3d", t, False))

        # 10. as_strided 重叠 (元素有重叠，non-overlapping=False)
        t = torch.randn(12).as_strided((3, 3), (4, 1))
        tensors.append(("as_strided_overlap", t, False))

        return tensors

    def _verify_preserve_format_result(self, src, result, scenario_name):
        """验证 preserve_format 的基本正确性"""
        # 数值一致
        self.assertEqual(src, result, atol=0, rtol=0,
                         msg=f"{scenario_name}: 数值不一致")
        # shape 一致
        self.assertEqual(list(src.shape), list(result.shape),
                         msg=f"{scenario_name}: shape 不一致")

    def _verify_stride_preserved(self, src, result, scenario_name, is_nod):
        """
        验证 stride 行为:
          - is_non_overlapping_and_dense=True → stride 应被保留
          - is_non_overlapping_and_dense=False → stride 可能不同(降级为 contiguous)
        """
        if is_nod:
            self.assertEqual(src.stride(), result.stride(),
                             msg=f"{scenario_name}: stride 未被保留 (is_non_overlapping_and_dense=True)")
        else:
            # 降级为 suggest_memory_format，结果可能是 contiguous 的
            # 不强求 stride 一致，但 shape 必须一致
            pass

    # ================================================================
    # H2D 场景: CPU → NPU
    # ================================================================

    def test_h2d_transpose_2d(self):
        """H2D: transpose 2D 非连续 → preserve_format 保留 stride"""
        cpu_t = torch.randn(4, 6).t()
        self.assertFalse(cpu_t.is_contiguous())
        self.assertTrue(cpu_t.is_non_overlapping_and_dense())

        npu_t = cpu_t.to("npu", memory_format=torch.preserve_format)
        self._verify_preserve_format_result(cpu_t, npu_t, "H2D-transpose_2d")
        self.assertEqual(str(npu_t.device), "npu:0")
        self.assertEqual(cpu_t.stride(), npu_t.stride(),
                         "H2D-transpose_2d: stride 未保留")
