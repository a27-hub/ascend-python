    def test_npu_quant_matmul_float8_e4m3fn_pertensor(self):
        torch.manual_seed(0)
        x1 = torch.randint(-5, 5, (8192, 320)).to(torch.float8_e4m3fn)
        x2 = torch.randint(-5, 5, (320, 2560)).to(torch.float8_e4m3fn)
        x1_clone = x1.clone()
        x2_clone = x2.clone()
        scale = torch.randn(1, dtype=torch.float32)
        scale_clone = scale.clone()
        supported_output = torch.matmul(x1.to(torch.float32), x2.to(torch.float32)) * scale
        custom_output = torch_npu.npu_quant_matmul(x1_clone.npu(), x2_clone.npu(), scale_clone.npu(), output_dtype=torch.float16)
        self.assertRtolEqual(supported_output.float().cpu().numpy(), custom_output.float().cpu().numpy(), 0.001)
